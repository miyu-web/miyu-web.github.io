Unity面部阴影SDF实现

更新于2026-03-20
|总字数:3.2k|阅读时长:14分钟|浏览量:

参考文章： https://zhuanlan.zhihu.com/p/670837192

前言

 对于角色面便阴影来说，有写实光照的软阴影与二分光照的硬阴影，对于写实光照的软阴影来说，基本上不存在面部阴影修正的问题，物理的阴影表现更符合我们的习惯，面部阴影的问题只可能会出现在强调边缘刻画的二分阴影中。

伦勃朗光（Rembrandt lighting）

 参考文章： https://zhuanlan.zhihu.com/p/411188212
 这是一种常见于肖像摄影的照明布局方式，最主要的特征是强调面部明暗对比，即高反差的光比。而卡通表现中经常使用到的是这种布光方式产生的阴影轮廓，这种轮廓最典型的特征就是一侧脸颊上产生的三角形的亮区。

解决面部阴影二分问题的方法

修改面部法线

 最经典的游戏就是罪恶都市，如下图，左边是未修改法线，右边是修改法线的效果：

包裹体法线

 这种方式主要使用球包裹体或结构简单的包裹体法线去映射到面部法线上，使面部显示出对应包裹体的阴影形状。

 但这样法线就取决于传递的是什么包裹体，光照阴影计算也会按照包裹体的法线进行计算，这种方式操作简单，效果也简单就是包裹体的阴影，支持的场景比较有限。

模型拓扑+发现修改

 把模型拓扑也改成对应阴影形状的拓扑，在此基础上调整法线，从而得到拓扑结构中的阴影形状。

 这种方式可以调整出理想的形状，但是非常耗时繁琐，且这种方式会导致阴影产生跳变。

法线贴图

 绘制法线贴图方式修改法线

距离场SDF


 Signed Distance Field（有向距离场）的应用十分广泛，常见于字体渲染与RayMarching中。这种方法的思路就是使用距离场记录光照数据，之后在根据光源方向取出对应的阴影形状，它的优势很明显可以完全自定义阴影形态，同时在不同形态之间拥有自然的过渡。其核心是需要一张距离场贴图，计算时候需要面部向前的向量，面部向左的向量和光照方向。距离场贴图只记录了当方向光照，所以计算另一方向的光照需要反转uv进行采样。

什么是SDF

参考文章： https://zhuanlan.zhihu.com/p/337944099
参考文章： https://zhuanlan.zhihu.com/p/279334552
 在实现面部SDF阴影之前，我们需要先了解一下什么SDF。
 SDF（signed distance field），中文名为有向距离场，SDF有2D和3D的区别，它的定义非常简单：每个像素（体素）记录自己与距离自己最近物体之间的距离，如果在物体内，则距离为负，正好在物体边界上则为0。
 如下图，我们规定白色为物体，黑色表示空。

 那么这副图像对应的SDF应该如下：

 为了图片显示，将有向距离值映射到了[0-1]范围，0.5表示物体内外的分界线。可以清楚看到，位于圆中心的点是最黑的，因为它处于物体最内部，而图片的四个角最白，因为它们距离圆最远。
 我们可以看到，当我们调节不同阈值的时候，SDF贴图就会显示不同的阈值区分：

实现SDF图像生成算法（8ssedt）

参考链接 https://zhuanlan.zhihu.com/p/337944099
 8SSEDT 通过4 次全局扫描，利用8 个相邻点的已知距离递推当前点的最小距离。

#define WIDTH  256
#define HEIGHT 256

struct Point
{
    int dx, dy;

    int DistSq() const { return dx*dx + dy*dy; }
};

struct Grid
{
    Point grid[HEIGHT][WIDTH];
};

Point inside = { 0, 0 };
Point empty = { 9999, 9999 };
Grid grid1, grid2;

Point Get( Grid &g, int x, int y )
{
    // OPTIMIZATION: you can skip the edge check code if you make your grid 
    // have a 1-pixel gutter.
    if ( x >= 0 && y >= 0 && x < WIDTH && y < HEIGHT )
        return g.grid[y][x];
    else
        return empty;
}

void Put( Grid &g, int x, int y, const Point &p )
{
    g.grid[y][x] = p;
}

void Compare( Grid &g, Point &p, int x, int y, int offsetx, int offsety )
{
    Point other = Get( g, x+offsetx, y+offsety );
    other.dx += offsetx;
    other.dy += offsety;

    if (other.DistSq() < p.DistSq())
        p = other;
}

void GenerateSDF( Grid &g )
{
    // Pass 0
    for (int y=0;y<HEIGHT;y++)
    {
        for (int x=0;x<WIDTH;x++)
        {
            Point p = Get( g, x, y );
            Compare( g, p, x, y, -1,  0 );
            Compare( g, p, x, y,  0, -1 );
            Compare( g, p, x, y, -1, -1 );
            Compare( g, p, x, y,  1, -1 );
            Put( g, x, y, p );
        }

        for (int x=WIDTH-1;x>=0;x--)
        {
            Point p = Get( g, x, y );
            Compare( g, p, x, y, 1, 0 );
            Put( g, x, y, p );
        }
    }

    // Pass 1
    for (int y=HEIGHT-1;y>=0;y--)
    {
        for (int x=WIDTH-1;x>=0;x--)
        {
            Point p = Get( g, x, y );
            Compare( g, p, x, y,  1,  0 );
            Compare( g, p, x, y,  0,  1 );
            Compare( g, p, x, y, -1,  1 );
            Compare( g, p, x, y,  1,  1 );
            Put( g, x, y, p );
        }

        for (int x=0;x<WIDTH;x++)
        {
            Point p = Get( g, x, y );
            Compare( g, p, x, y, -1, 0 );
            Put( g, x, y, p );
        }
    }
}

private void GenerateSDFGrid(Grid grid, int w, int h)
{
    // Pass 0: 从上到下，从左到右
    for (int y = 0; y < h; y++)
    {
        for (int x = 0; x < w; x++)
        {
            Point p = grid.Get(x, y);
            Compare(grid, ref p, x, y, -1, 0);
            Compare(grid, ref p, x, y, 0, -1);
            Compare(grid, ref p, x, y, -1, -1);
            Compare(grid, ref p, x, y, 1, -1);
            grid.Set(x, y, p);
        }
        for (int x = w - 1; x >= 0; x--)
        {
            Point p = grid.Get(x, y);
            Compare(grid, ref p, x, y, 1, 0);
            grid.Set(x, y, p);
        }
    }

    // Pass 1: 从下到上，从右到左
    for (int y = h - 1; y >= 0; y--)
    {
        for (int x = w - 1; x >= 0; x--)
        {
            Point p = grid.Get(x, y);
            Compare(grid, ref p, x, y, 1, 0);
            Compare(grid, ref p, x, y, 0, 1);
            Compare(grid, ref p, x, y, -1, 1);
            Compare(grid, ref p, x, y, 1, 1);
            grid.Set(x, y, p);
        }
        for (int x = 0; x < w; x++)
        {
            Point p = grid.Get(x, y);
            Compare(grid, ref p, x, y, -1, 0);
            grid.Set(x, y, p);
        }
    }
}

private void Compare(Grid grid, ref Point p, int x, int y, int dx, int dy)
{
    Point other = grid.Get(x + dx, y + dy);
    if (other.IsEmpty) return;
    other.dx += dx;
    other.dy += dy;
    if (other.DistSq < p.DistSq)
        p = other;
}


// 点结构
[System.Serializable]
private struct Point
{
    public int dx, dy;
    public int DistSq => dx * dx + dy * dy;
    public bool IsEmpty => dx == 9999 && dy == 9999;

    public static Point Empty => new Point(9999, 9999);

    public Point(int dx, int dy)
    {
        this.dx = dx;
        this.dy = dy;
    }
}

// Grid 类
private class Grid
{
    public Point[] data;
    private int width, height;

    public Grid(int w, int h, bool inside)
    {
        width = w;
        height = h;
        data = new Point[w * h];
        Point init = inside ? new Point(0, 0) : Point.Empty;
        for (int i = 0; i < data.Length; i++)
            data[i] = init;
    }

    public Point Get(int x, int y)
    {
        if (x < 0 || x >= width || y < 0 || y >= height)
            return Point.Empty;
        return data[y * width + x];
    }

    public void Set(int x, int y, Point p)
    {
        if (x < 0 || x >= width || y < 0 || y >= height)
            return;
        data[y * width + x] = p;
    }
}
private void GenerateAndSave(Texture2D sourceTex)
{
    Texture2D readableTex = DuplicateTexture(sourceTex);
    Color32[] pixels = readableTex.GetPixels32();
    int w = readableTex.width;
    int h = readableTex.height;

    Grid insideGrid = new Grid(w, h, true);
    Grid outsideGrid = new Grid(w, h, false);

    for (int y = 0; y < h; y++)
    {
        for (int x = 0; x < w; x++)
        {
            Color32 c = pixels[y * w + x];
            float brightness = (0.299f * c.r + 0.587f * c.g + 0.114f * c.b) / 255f;
            bool inside = brightness > threshold;

            if (inside)
            {
                insideGrid.Set(x, y, new Point(0, 0));
                outsideGrid.Set(x, y, Point.Empty);
            }
            else
            {
                outsideGrid.Set(x, y, new Point(0, 0));
                insideGrid.Set(x, y, Point.Empty);
            }
        }
    }

    GenerateSDFGrid(insideGrid, w, h);
    GenerateSDFGrid(outsideGrid, w, h);

    float[] sdf = new float[w * h];
    float maxDist = spread;
    for (int i = 0; i < w * h; i++)
    {
        float dIn = Mathf.Sqrt(insideGrid.data[i].DistSq);
        float dOut = Mathf.Sqrt(outsideGrid.data[i].DistSq);
        float dist = dIn - dOut;
        sdf[i] = Mathf.Clamp(dist / maxDist, -1f, 1f);
    }

    string baseName = Path.GetFileNameWithoutExtension(sourceTex.name);


    string pngPath = Path.Combine(outputFolder, baseName + "_sdf.png");
    SaveAsPNG(sdf, w, h, pngPath);

}
private void SaveAsPNG(float[] sdf, int width, int height, string absolutePath)
{
    string directory = Path.GetDirectoryName(absolutePath);
    if (!Directory.Exists(directory))
        Directory.CreateDirectory(directory);

    Texture2D tex = new Texture2D(width, height, TextureFormat.R8, false);
    Color[] colors = new Color[sdf.Length];
    for (int i = 0; i < sdf.Length; i++)
    {
        float v = (sdf[i] + 1f) / 2f;
        colors[i] = new Color(v, v, v, 1);
    }
    tex.SetPixels(colors);
    tex.Apply();

    byte[] pngData = tex.EncodeToPNG();
    File.WriteAllBytes(absolutePath, pngData);
}

实现水平面部SDF阴影

half sdfRight=SAMPLE_TEXTURE2D(_SDFlMap, sampler_SDFlMap, input.uv).r;
half sdfLeft=SAMPLE_TEXTURE2D(_SDFlMap, sampler_SDFlMap, float2(1.0-input.uv.x,input.uv.y)).r;

half3 front_dir= unity_ObjectToWorld._13_23_33;
half3 right_dir = unity_ObjectToWorld._11_21_31;
half3 up_dir=unity_ObjectToWorld._12_22_32;
half fl=dot(front_dir.xz,mainLight.direction.xz);
half rl=dot(right_dir.xz,mainLight.direction.xz);
half ul=dot(up_dir,mainLight.direction);

half sdfSide=lerp(sdfLeft,sdfRight,step(0,rl));
half isInSideShaow=step(sdfSide,fl*0.5+0.5);
color.rgb*=lerp(_ShadowColor.rgb,1.0,isInSideShaow);

来详细解析一下代码：

  • 首先我们的贴图是只存放了一个方向的光照SDF信息，所以采样光从左边照射过来的时候需要反向uv采样。
  • unity_ObjectToWorld是Unity内置的矩阵变量，其核心作用是将模型空间的点或者向量转换成世界空间的点或者坐标。矩阵完全遵守以下结构：
    • 列 1（_11_21_31）：物体本地 X 轴（右向）在世界空间的方向向量。
    • 列 2（_12_22_32）：物体本地 Y 轴（上向）在世界空间的方向向量。
    • 列 3（_13_23_33）：物体本地 Z 轴（前向）在世界空间的方向向量。
  • 获取方向后进行点乘，点乘我们有一种意义就是判断方向a与方向b的重合度，这样我们就很好理解三个点乘的作用。
    • fl：判断光线从前面来还是从后面来的，0正好是正面和背面的分割线，如下图：
    • rl：判断光线从左面来还是从右面来的，0正好是左面和右面的分割线，如下图：
    • ul：判断光线从上面来还是从下面来的，0正好是上面和下面的分割线，这个会用在360度面部SDF阴影上，水平的其实不用考虑。
  • 根据光线是从左面照射过来还是从右面照射过来决定采样。
  • 比较采样的贴图与当前前向照射的阈值进行判断，因为我们先判断了光线从左还是从右采样，然后需要判断光线是从前还是从后，因为光线在前后分界线处开始就会处在阴影里的。如果小于阈值那么就返回0，表示在阴影中，如果大于阈值则返回1，表示不在阴影里。（因为SDF贴图中表示的是不同角度光线下面部阴影阈值）

实现360度面部SDF阴影

参考链接：https://zhuanlan.zhihu.com/p/670837192
模型资源链接: <https://pan.baidu.com/s/1Axb2kWqHvE-5NZoDjtLAzw?pwd=m68j 提取码: m68j>
其中是我对参考链接中提供的模型资源进行的导出和设置，可以直接使用。

Light mainLight = GetMainLight();
float3 L=normalize(mainLight.direction);

half4 finalColor = SAMPLE_TEXTURE2D(_MainTex, sampler_MainTex, i.uv);

half3 right_dir = normalize(unity_ObjectToWorld._11_21_31);
half3  up_dir= normalize(unity_ObjectToWorld._12_22_32);
half3 front_dir = normalize(unity_ObjectToWorld._13_23_33);

float isLeftOrRight = step(0, dot(right_dir,L));
float side_x = lerp(1- i.uv.x,i.uv.x, isLeftOrRight);
float2 uv_00 = float2(side_x,i.uv.y)/9;

float cos_theta = dot(front_dir, L);//-1~1
float theta = cos_theta*0.5+0.5;//0~1
float linear_theta = 8.0 * theta; //0~8

float cos_phi = dot(up_dir, L);//-1~1
float phi = cos_phi*0.5+0.5;//0~1
float linear_phi = 8.0 * phi; //0~8

float current_row = floor(linear_theta);
float current_row_1 = current_row + 1; 
float theta_weight = frac(linear_theta);

float current_col = floor(linear_phi);
float current_col_1 = current_col + 1;
float phi_weight = frac(linear_phi);


float2 uv_1 = uv_00+min(float2(current_row,current_col),8)/9;
float2 uv_2 = uv_00+min(float2(current_row_1,current_col),8)/9;
float2 uv_3 = uv_00+min(float2(current_row,current_col_1),8)/9;
float2 uv_4 = uv_00+min(float2(current_row_1,current_col_1),8)/9;

half sdf_1 = SAMPLE_TEXTURE2D(_SDFMap, sampler_SDFMap, uv_1).r;
half sdf_2 = SAMPLE_TEXTURE2D(_SDFMap, sampler_SDFMap, uv_2).r;
half sdf_3 = SAMPLE_TEXTURE2D(_SDFMap, sampler_SDFMap, uv_3).r;
half sdf_4 = SAMPLE_TEXTURE2D(_SDFMap, sampler_SDFMap, uv_4).r;


half shadow_h1 = lerp(sdf_1, sdf_2, theta_weight);
half shadow_h2 = lerp(sdf_3, sdf_4, theta_weight);

half final_shadow = step(0.5,lerp(shadow_h1, shadow_h2, phi_weight));
finalColor.rgb *= lerp(1.0,_ShadowColor.rgb,final_shadow);

其实里面的逻辑也很简单，跟水平SDF实现逻辑是一样的，只不过是多了一个维度，对水平和垂直SDF进行双线性插值。

在测试里我也使用了参考文章中的图集和模型，在Unity中可以设置成下图来减小带宽：

因为图集中的光照是从模型右面开始照射的，所以是float isLeftOrRight = step(0, dot(right_dir.xz,L.xz));
cos_theta与cos_phi是水平夹角与垂直夹角的余弦，进行归一化到区间[0,1]，乘以$8$则是因为在uv0基础上，最大x与y都是八格。
floor函数可以获取当前行数和列数，frac函数则是可以获取每个格子中的进度，依次作为双线性插值中的权重。
