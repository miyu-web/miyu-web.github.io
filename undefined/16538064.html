详解Blender着色器节点-漫反射BSDF

更新于2026-04-25
|总字数:4.4k|阅读时长:20分钟|浏览量:

前言

我们知道，着色器节点在Blender中是不可或缺的，不同的着色器节点可以实现不同的视觉效果，而这种不同之处就取决于它们底层实现的算法不同，此篇将以漫反射BSDF节点开始，一步一步探索Blender渲染背后的秘密。

什么是BSDF

BSDF，全称Bidirectional scattering distribution function，即双向散射分布函数，它是由双向反射分布函数BRDF和双向透射分布函数BTDF共同构成。

漫反射BSDF算法

Blender漫反射BSDF是由什么算法构成的？

先来看一下软件内的节点：

可以看到节点有一个颜色属性，一个糙度属性，一个法向属性。当我们把鼠标放到节点上时，就会出现以下提示：

我们去到在线手册，也能看到官方对于漫反射BSDF节点的解释：

  • 漫反射BSDF节点是用于添加Lambertian和Oren-Nayar漫反射的。
  • 节点中的糙度只对Cycles渲染器起作用，对于EEVEE渲染器并无作用。
  • 当渲染器为Cycles时，糙度为0时给出标准的Lambertian反射，当不为0的时候，则会激活Oren-Nayar。

来看一下对比图：


可以看到，当渲染器为EEVEE时候，糙度参数并不会对效果产生影响，默认就只是输出Lambertian反射。
而当渲染为Cycles时才会产生变化，并且当糙度为0的时候，效果和EEVEE渲染效果是一致的，也就是只输出Lambertian反射。

而在Blender常用的原理化BSDF着色器节点中，在漫射中也拥有一个糙度：漫射粗糙度。

这一个糙度与金属度下方的糙度不一样，这一个糙度决定了漫反射的计算方式，在官方文档中也有解释：

而金属度下方的糙度是控制镜面反射与透射的，对漫反射粗糙控制并无作用：

其中提及当粗糙度为1的时候就会是漫反射，那么此时才会回到我们之前说的逻辑之上，由漫射粗糙度来决定最终效果，否则便是混合效果。

说了这么多，那么我们就开始来查看底层是如何计算的。

Cycles渲染器中的算法

Lambertian反射

这个实际上就很简单了，就是兰伯特模型，这是最经典的漫反射光照模型。由此可见兰伯特模型在PBR物理计算中也会使用，PBR也不是那么完全按照物理公式去计算的。

half3 lambert = max(0.0,dot(n,l))*albedo*1/PI;

为什么要除以PI呢？这是为了符合能量守恒定律，更贴近真实物理性质。
而在OSL语言中，从设计上，这个语言是没用光照数据的，因此我们很难手写兰伯特光照，只能使用内置的diffuse闭包函数，不要问为什么，因为这个语言起初的设计就是这样的，这个语言的设计思路就是只是描述这个材质的属性，所以并不会设计到光，因此你要在Cycles渲染器中获取光线方向，颜色这类数据，你只能在渲染器中自行改造并传递正确的值给OSL的全局参数当中去。
官方文档链接：https://open-shading-language.readthedocs.io/en/v1.15.3.0/syntax.html
因此我们要实现一个兰伯特光照，除非定义伪光，要么就只能使用闭包函数，或者你有能力就去修改渲染器逻辑：

shader m_lambert(color Color=1.0,normal Normal=N,output closure color finalColor=0)
{
    finalColor=diffuse(Normal)*Color;
}

这个方法官方文档有说是废弃的，将在版本2.0的OSL被删除，可以用：

shader m_lambert(color Color=1.0,normal Normal=N,output closure color finalColor=0)
{
    finalColor=oren_nayar_diffuse_bsdf(Normal,Color,0);
}

自定义的兰伯特着色代码：

shader m_lambert(color Color=1.0,normal Normal=N,output  color finalColor=0)
{
  //此定义是一个伪光
    vector L = vector(0,0,1);
    finalColor =max(0.0,dot(Normal,L))*Color*0.3183098861837067;
}
🐍 为什么使用自定义的兰伯特着色代码，对比糙度为0的漫反射BSDF，效果不一样呢？

这是因为存在环境光的影响，如果要对比，我们需要关闭场景世界，同时将场景世界的强度改为0，不透明度改为0。

可以看见确实是很简单的兰伯特模型。

ccl_device int bsdf_diffuse_setup(ccl_private DiffuseBsdf *bsdf)
{
  bsdf->type = CLOSURE_BSDF_DIFFUSE_ID;
  return SD_BSDF | SD_BSDF_HAS_EVAL;
}

ccl_device Spectrum bsdf_diffuse_eval(const ccl_private ShaderClosure *sc,
                                      const float3 wi,
                                      const float3 wo,
                                      ccl_private float *pdf)
{
  const ccl_private DiffuseBsdf *bsdf = (const ccl_private DiffuseBsdf *)sc;
  const float3 N = bsdf->N;

  const float cosNO = fmaxf(dot(N, wo), 0.0f) * M_1_PI_F;
  *pdf = cosNO;
  return make_spectrum(cosNO);
}

ccl_device int bsdf_diffuse_sample(const ccl_private ShaderClosure *sc,
                                   const float3 Ng,
                                   const float3 wi,
                                   const float2 rand,
                                   ccl_private Spectrum *eval,
                                   ccl_private float3 *wo,
                                   ccl_private float *pdf)
{
  const ccl_private DiffuseBsdf *bsdf = (const ccl_private DiffuseBsdf *)sc;
  const float3 N = bsdf->N;

  // distribution over the hemisphere
  sample_cos_hemisphere(N, rand, wo, pdf);

  if (dot(Ng, *wo) > 0.0f) {
    *eval = make_spectrum(*pdf);
  }
  else {
    *pdf = 0.0f;
    *eval = zero_spectrum();
  }
  return LABEL_REFLECT | LABEL_DIFFUSE;
}

/* TRANSLUCENT */

ccl_device int bsdf_translucent_setup(ccl_private DiffuseBsdf *bsdf)
{
  bsdf->type = CLOSURE_BSDF_TRANSLUCENT_ID;
  return SD_BSDF | SD_BSDF_HAS_EVAL | SD_BSDF_HAS_TRANSMISSION;

Oren-Nayar反射

这是一种比朗伯法更为复杂的扩散计算，它提供了一个粗糙度指数来控制扩散贡献的扩展，通常它的颜色是材料的主要表面颜色，这种照明模型通常用于没有高光贡献的材料，如粘土或粉笔或哑光涂料。
Oren-Nayar 模型的核心思想基于微平面模型：它将任何肉眼看起来平滑的表面，在微观层面想象成由大量朝向各异的微小平面(V-shaped facets)组成。这些小微平面本身被认为是完美的。
下方是一个由GLSL实现的拟合曲线代码，使用拟合曲线的原因在于Oren-Nayar是基于微表面理论的，这样的计算很复杂，类似于PBR，都会采用拟合曲线取尽可能模拟来达到减小计算性能的目的。

float OrenNayar( in vec3 l, in vec3 n, in vec3 v, float r ) 
{ 
    float r2 = r*r; 
    float a = 1.0 - 0.5*(r2/(r2+0.57)); 
    float b = 0.45*(r2/(r2+0.09)); 
    float nl = dot(n, l); 
    float nv = dot(n, v); 
    float ga = dot(v - n*nv, n - n*nl); 
    return max(0.0, nl) * (a + b*max(0.0,ga) * sqrt((1.0-nv*nv)*(1.0-nl*nl)) / max(nl, nv));
}

而在blender中，其计算如下：

/* SPDX-FileCopyrightText: 2011-2022 Blender Foundation
 *
 * SPDX-License-Identifier: Apache-2.0 */

#pragma once

#include "kernel/types.h"

#include "kernel/sample/mapping.h"

CCL_NAMESPACE_BEGIN

struct OrenNayarBsdf {
  SHADER_CLOSURE_BASE;

  float roughness;
  float a;
  float b;
  Spectrum multiscatter_term;
};

static_assert(sizeof(ShaderClosure) >= sizeof(OrenNayarBsdf), "OrenNayarBsdf is too large!");

/* NOTE: This implements the improved Oren-Nayar model by Yasuhiro Fujii
 * (https://mimosa-pudica.net/improved-oren-nayar.html), plus an
 * energy-preserving multi-scattering term based on the OpenPBR specification
 * (https://academysoftwarefoundation.github.io/OpenPBR). */

ccl_device_inline float bsdf_oren_nayar_G(const float cosTheta)
{
  if (cosTheta < 1e-6f) {
    /* The tan(theta) term starts to act up at low cosTheta, so fall back to Taylor expansion. */
    return (M_PI_2_F - 2.0f / 3.0f) - cosTheta;
  }
  const float sinTheta = sin_from_cos(cosTheta);
  const float theta = safe_acosf(cosTheta);
  return sinTheta * (theta - 2.0f / 3.0f - sinTheta * cosTheta) +
         2.0f / 3.0f * (sinTheta / cosTheta) * (1.0f - sqr(sinTheta) * sinTheta);
}

ccl_device Spectrum bsdf_oren_nayar_get_intensity(const ccl_private ShaderClosure *sc,
                                                  const float3 n,
                                                  const float3 v,
                                                  const float3 l)
{
  const ccl_private OrenNayarBsdf *bsdf = (const ccl_private OrenNayarBsdf *)sc;
  const float nl = max(dot(n, l), 0.0f);
  if (bsdf->b <= 0.0f) {
    return make_spectrum(nl * M_1_PI_F);
  }
  const float nv = max(dot(n, v), 0.0f);
  float t = dot(l, v) - nl * nv;

  if (t > 0.0f) {
    t /= max(nl, nv) + FLT_MIN;
  }

  const float single_scatter = bsdf->a + bsdf->b * t;

  const float El = bsdf->a * M_PI_F + bsdf->b * bsdf_oren_nayar_G(nl);
  const Spectrum multi_scatter = bsdf->multiscatter_term * (1.0f - El);

  return nl * (make_spectrum(single_scatter) + multi_scatter);
}

ccl_device int bsdf_oren_nayar_setup(const ccl_private ShaderData *sd,
                                     ccl_private OrenNayarBsdf *bsdf,
                                     const Spectrum color)
{
  bsdf->type = CLOSURE_BSDF_OREN_NAYAR_ID;

  const float sigma = saturatef(bsdf->roughness);
  bsdf->a = 1.0f / (M_PI_F + sigma * (M_PI_2_F - 2.0f / 3.0f));
  bsdf->b = sigma * bsdf->a;

  /* Compute energy compensation term (except for (1.0f - El) factor since it depends on wo). */
  const Spectrum albedo = saturate(color);
  const float Eavg = bsdf->a * M_PI_F + ((M_2PI_F - 5.6f) / 3.0f) * bsdf->b;
  const Spectrum Ems = M_1_PI_F * sqr(albedo) * (Eavg / (1.0f - Eavg)) /
                       (one_spectrum() - albedo * (1.0f - Eavg));
  const float nv = max(dot(bsdf->N, sd->wi), 0.0f);
  const float Ev = bsdf->a * M_PI_F + bsdf->b * bsdf_oren_nayar_G(nv);
  bsdf->multiscatter_term = Ems * (1.0f - Ev);

  return SD_BSDF | SD_BSDF_HAS_EVAL;
}

ccl_device Spectrum bsdf_oren_nayar_eval(const ccl_private ShaderClosure *sc,
                                         const float3 wi,
                                         const float3 wo,
                                         ccl_private float *pdf)
{
  const ccl_private OrenNayarBsdf *bsdf = (const ccl_private OrenNayarBsdf *)sc;
  const float cosNO = dot(bsdf->N, wo);
  if (cosNO > 0.0f) {
    *pdf = cosNO * M_1_PI_F;
    return bsdf_oren_nayar_get_intensity(sc, bsdf->N, wi, wo);
  }
  *pdf = 0.0f;
  return zero_spectrum();
}

ccl_device int bsdf_oren_nayar_sample(const ccl_private ShaderClosure *sc,
                                      const float3 Ng,
                                      const float3 wi,
                                      const float2 rand,
                                      ccl_private Spectrum *eval,
                                      ccl_private float3 *wo,
                                      ccl_private float *pdf)
{
  const ccl_private OrenNayarBsdf *bsdf = (const ccl_private OrenNayarBsdf *)sc;

  sample_cos_hemisphere(bsdf->N, rand, wo, pdf);

  if (dot(Ng, *wo) > 0.0f) {
    *eval = bsdf_oren_nayar_get_intensity(sc, bsdf->N, wi, *wo);
  }
  else {
    *pdf = 0.0f;
    *eval = zero_spectrum();
  }

  return LABEL_REFLECT | LABEL_DIFFUSE;
}

那我们结合上面的底层代码，来自己实现osl着色代码：

#define M_PI_F     3.1415926535897932             /* `pi` */
#define M_PI_2_F   1.5707963267948966             /* `pi/2` */
#define M_1_PI_F   0.3183098861837067             /* `1/pi` */
#define M_2PI_F    6.2831853071795864             /* `2*pi` */
#define FLT_MIN    1.175494350822287507969e-38

float saturate(float x) { return clamp(x, 0.0, 1.0); }
vector saturate(vector c)  { return clamp(c, 0.0, 1.0); }

float oren_nayar_G(float cosTheta)
{
    if(cosTheta<1e-6)
    {
        //当 cosTheta 很小时，tan(theta) 项开始失效，所以回退到泰勒展开
        return (M_PI_2_F-2.0/3.0)-cosTheta;
    }
    float sinTheta = sqrt(max(1.0-cosTheta*cosTheta,0.0));
    float theta = acos(clamp(cosTheta,-1.0,1.0));
    return sinTheta * (theta - 2.0/3.0 - sinTheta * cosTheta)+ 2.0/3.0 * (sinTheta / cosTheta) * (1.0 - sinTheta*sinTheta*sinTheta);
}

color oren_nayar_bsdf(vector albedo,normal n,vector v,vector l,float roughness)
{
    float nl=dot(n,l);

    if(nl<=0.0)
    {
      return 0.0;
    }
    nl=saturate(nl);

    float sigma = saturate(roughness);
    float a = 1.0/(M_PI_F+sigma*(M_PI_2_F-2.0/3.0));
    float b = sigma*a;
    
    if (b <= 0.0) 
    {
      return nl * M_1_PI_F;
    }


    float nv=max(dot(n,v),0.0);
    float Ev = a*M_PI_F+b*oren_nayar_G(nv);

    color albedo_sat = saturate(albedo);
    float Eavg = a*M_PI_F+(M_2PI_F-5.6)/3.0*b;

    vector Ems=M_1_PI_F*albedo_sat*albedo_sat*(Eavg/(1.0-Eavg))/(vector(1.0,1.0,1.0)-albedo_sat*(1.0-Eavg));

    vector multiscatter_term = Ems * (1.0 - Ev);
    

    float t = dot(l,v)-nl*nv;
    if(t>0.0)
    {
      t/=max(nl,nv)+FLT_MIN;
    }

    float single_scatter = a+b*t;
    float El = a * M_PI_F + b * oren_nayar_G(nl);
    vector multi_scatter = multiscatter_term * (1.0 - El);

    vector intensity = nl * (vector(single_scatter,single_scatter,single_scatter) + multi_scatter);
    return intensity;
}

shader m_oren_nayar(color Color=1.0,float roughness=0.0,normal Normal=N,output color finalColor=0)
{
    vector L = vector(0,0,1);
    finalColor=oren_nayar_bsdf(Color,Normal,I,L,roughness)*Color;
}

算法是没有什么问题的，在编辑器内关闭环境光影响时并且采用渲染模式是正常的，但是在最终渲染时默认的漫反射BSDF出图会比自定义的稍亮一些，原因是我虽然编辑器关闭了，但是渲染没有关闭，导致渲染的时候还是叠加了环境光，将背景改为黑色就可以了。
